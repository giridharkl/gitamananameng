\newcommand\Chapter[3]{
  \chapter[#1 #2]{#1\\\Large{#2\\ {\vspace{-5mm}\normalsize(#3)}}}
  \vspace{-10mm}
  \newpgfornamentfamily{vectorian} %design
  \begin{center}
    
    \pgfornament[anchor=center,ydelta=0pt,width=5cm, color=\chapBeginSymColor]{84}
  \end{center}
  %print different header for each chapter
  \fancyhead[RO]{\footnotesize \arialUC Gita Mananam} %Odd page Right
  \fancyhead[LO]{\footnotesize #1} % Odd page Left
  \fancyhead[LE]{\footnotesize \arialUC Gita Mananam} %Even page Left
  \fancyhead[RE]{\footnotesize #1} %Even page Right
}
